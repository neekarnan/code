import torch
import torch.nn as nn
import torch.nn.functional as F
from torchvision import datasets, transforms
from torch.utils.data import DataLoader
import matplotlib.pyplot as plt
import numpy as np

# 1. Load MNIST dataset
transform = transforms.Compose([
    transforms.ToTensor(),
    transforms.Lambda(lambda x: x.view(-1))
])

data = datasets.MNIST("./data", train=True,
                      download=True, transform=transform)
loader = DataLoader(data, batch_size=128, shuffle=True)

# 2. VAE Model
class VAE(nn.Module):
    def __init__(self):
        super().__init__()

        # Encoder
        self.fc1 = nn.Linear(784, 256)
        self.fc2 = nn.Linear(256, 64)
        self.mean = nn.Linear(64, 2)
        self.logvar = nn.Linear(64, 2)

        # Decoder
        self.fc3 = nn.Linear(2, 64)
        self.fc4 = nn.Linear(64, 256)
        self.fc5 = nn.Linear(256, 784)

    def encode(self, x):
        x = F.relu(self.fc1(x))
        x = F.relu(self.fc2(x))
        return self.mean(x), self.logvar(x)

    def reparameterize(self, mu, logvar):
        std = torch.exp(0.5 * logvar)
        eps = torch.randn_like(std)
        return mu + std * eps

    def decode(self, z):
        z = F.relu(self.fc3(z))
        z = F.relu(self.fc4(z))
        return torch.sigmoid(self.fc5(z))

    def forward(self, x):
        mu, logvar = self.encode(x)
        z = self.reparameterize(mu, logvar)
        return self.decode(z), mu, logvar

# 3. Loss Function
def loss_fn(output, x, mu, logvar):
    recon = F.binary_cross_entropy(output, x, reduction="sum")
    kl = -0.5 * torch.sum(
        1 + logvar - mu.pow(2) - logvar.exp()
    )
    return recon + kl

# 4. Training
device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
model = VAE().to(device)
optimizer = torch.optim.Adam(model.parameters(), lr=0.001)

for epoch in range(15):
    total_loss = 0

    for x, _ in loader:
        x = x.to(device)

        output, mu, logvar = model(x)
        loss = loss_fn(output, x, mu, logvar)

        optimizer.zero_grad()
        loss.backward()
        optimizer.step()

        total_loss += loss.item()

    print("Epoch:", epoch + 1,
          "Loss:", total_loss / len(data))

# 5. Generate 15 x 15 digit grid
model.eval()
n = 15
figure = np.zeros((28 * n, 28 * n))
grid = np.linspace(-3, 3, n)

with torch.no_grad():
    for i, y in enumerate(grid[::-1]):
        for j, x in enumerate(grid):
            z = torch.tensor([[x, y]], dtype=torch.float32).to(device)
            digit = model.decode(z).cpu().reshape(28, 28).numpy()
            figure[i*28:(i+1)*28, j*28:(j+1)*28] = digit

plt.figure(figsize=(8, 8))
plt.imshow(figure, cmap="gray")
plt.title("Generated Digits")
plt.axis("off")
plt.show()

# 6. Latent Space Visualization
points = []
labels = []

with torch.no_grad():
    for x, y in loader:
        mu, _ = model.encode(x.to(device))
        points.append(mu.cpu().numpy())
        labels.append(y.numpy())

points = np.concatenate(points)
labels = np.concatenate(labels)

plt.figure(figsize=(8, 6))
plt.scatter(points[:, 0], points[:, 1],
            c=labels, cmap="tab10", s=2)
plt.colorbar(label="Digit")
plt.xlabel("Latent Dimension 1")
plt.ylabel("Latent Dimension 2")
plt.title("VAE Latent Space")
plt.show()
