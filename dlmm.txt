def lab2_build_model(hidden_layers=1, neurons=128, activation="relu",
                     optimizer_name="adam", lr=0.001, batch_norm=False):
    model = Sequential()
    model.add(Flatten(input_shape=(28, 28)))

    for _ in range(hidden_layers):
        model.add(Dense(neurons, activation=activation))
        if batch_norm:
            model.add(BatchNormalization())

    model.add(Dense(10, activation="softmax"))

    if optimizer_name == "adam":
        opt = Adam(learning_rate=lr)
    elif optimizer_name == "sgd":
        opt = SGD(learning_rate=lr)
    else:
        raise ValueError("optimizer_name must be 'adam' or 'sgd'")

    model.compile(
        optimizer=opt,
        loss="sparse_categorical_crossentropy",
        metrics=["accuracy"]
    )
    return model


def lab2_train(model, x_train, y_train, x_val, y_val,
               epochs=10, batch_size=128):
    start = time.time()
    history = model.fit(
        x_train, y_train,
        epochs=epochs,
        batch_size=batch_size,
        validation_data=(x_val, y_val),
        verbose=1
    )
    elapsed = time.time() - start
    return history, elapsed


def lab2_plot_loss(histories, title):
    plt.figure(figsize=(9, 5))
    for name, h in histories.items():
        plt.plot(h.history["loss"], marker="o", label=name)
    plt.xlabel("Epoch")
    plt.ylabel("Training Loss")
    plt.title(title)
    plt.legend()
    plt.grid(True)
    plt.show()

import torch

print("\nLAB 3 DEVICE:", l3_device)

    transforms.ToTensor(),
    transforms.Normalize((0.5,), (0.5,)),
    transforms.Lambda(lambda x: x.view(-1))

class ShallowMLP(nn.Module):
    def __init__(self):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(784, 128),
            nn.ReLU(),
            nn.Linear(128, 10)
        )

    def forward(self, x):
        return self.net(x)


class DeepMLP(nn.Module):
    def __init__(self, width=128, dropout=0.3):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(784, width),
            nn.BatchNorm1d(width),
            nn.ReLU(),
            nn.Dropout(dropout),

            nn.Linear(width, width),
            nn.BatchNorm1d(width),
            nn.ReLU(),
            nn.Dropout(dropout),

            nn.Linear(width, width),
            nn.BatchNorm1d(width),
            nn.ReLU(),
            nn.Dropout(dropout),

            nn.Linear(width, width),
            nn.BatchNorm1d(width),
            nn.ReLU(),
            nn.Dropout(dropout),

            nn.Linear(width, 10)
        )

    def forward(self, x):
        return self.net(x)


def l3_train_model(model, train_loader, val_loader, epochs=10,
                    lr=0.001, optimizer_name="adam"):
    criterion = nn.CrossEntropyLoss()

    if optimizer_name == "adam":
        optimizer = optim.Adam(model.parameters(), lr=lr)
    elif optimizer_name == "sgd":
        optimizer = optim.SGD(model.parameters(), lr=lr, momentum=0.9)
    else:
        raise ValueError("optimizer_name must be 'adam' or 'sgd'")

    history = {
        "train_loss": [], "val_loss": [],
        "train_acc": [], "val_acc": [],
        "epoch_time": []
    }

    total_start = time.time()

    for epoch in range(epochs):
        epoch_start = time.time()
        model.train()
        train_loss = 0.0
        train_correct = 0
        train_total = 0

        for x, y in train_loader:
            x, y = x.to(l3_device), y.to(l3_device)
            optimizer.zero_grad()
            out = model(x)
            loss = criterion(out, y)
            loss.backward()
            optimizer.step()

            train_loss += loss.item() * x.size(0)
            train_correct += (out.argmax(1) == y).sum().item()
            train_total += y.size(0)

        model.eval()
        val_loss = 0.0
        val_correct = 0
        val_total = 0

        with torch.no_grad():
            for x, y in val_loader:
                x, y = x.to(l3_device), y.to(l3_device)
                out = model(x)
                loss = criterion(out, y)
                val_loss += loss.item() * x.size(0)
                val_correct += (out.argmax(1) == y).sum().item()
                val_total += y.size(0)

        history["train_loss"].append(train_loss / train_total)
        history["val_loss"].append(val_loss / val_total)
        history["train_acc"].append(train_correct / train_total)
        history["val_acc"].append(val_correct / val_total)
        history["epoch_time"].append(time.time() - epoch_start)

        print(
            f"Epoch [{epoch+1}/{epochs}] "
            f"Train Loss: {history['train_loss'][-1]:.4f} "
            f"Train Acc: {history['train_acc'][-1]:.4f} "
            f"Val Loss: {history['val_loss'][-1]:.4f} "
            f"Val Acc: {history['val_acc'][-1]:.4f}"
        )

    history["training_time"] = time.time() - total_start
    return history


def l3_evaluate(model, loader):
    model.eval()
    correct = 0
    total = 0
    with torch.no_grad():
        for x, y in loader:
            x, y = x.to(l3_device), y.to(l3_device)
            correct += (model(x).argmax(1) == y).sum().item()
            total += y.size(0)
    return correct / total


def l3_count_params(model):
    return sum(p.numel() for p in model.parameters() if p.requires_grad)

def l4_activation():
    if L4_ACTIVATION == "relu":
        return nn.ReLU()
    if L4_ACTIVATION == "leaky_relu":
        return nn.LeakyReLU()
    if L4_ACTIVATION == "sigmoid":
        return nn.Sigmoid()
    if L4_ACTIVATION == "tanh":
        return nn.Tanh()
    raise ValueError("Unsupported activation")


class ConfigurableCNN(nn.Module):
    def __init__(self):
        super().__init__()
        blocks = []
        in_channels = 1
        channels = [16, 32, 64, 128, 256]

        for i in range(L4_NUM_CONV_BLOCKS):
            out_channels = channels[i]
            blocks.append(
                nn.Conv2d(
                    in_channels,
                    out_channels,
                    kernel_size=L4_KERNEL_SIZE,
                    stride=L4_STRIDE,
                    padding=L4_PADDING
                )
            )
            if L4_USE_BATCHNORM:
                blocks.append(nn.BatchNorm2d(out_channels))
            blocks.append(l4_activation())

            # The exercise varies stride/kernel/padding. To keep the network
            # valid for aggressive settings such as stride=2, pool only when
            # the feature map can still be spatially downsampled.
            blocks.append(nn.MaxPool2d(kernel_size=2, stride=2, ceil_mode=True))
            in_channels = out_channels

        self.features = nn.Sequential(*blocks)

        with torch.no_grad():
            dummy = torch.zeros(1, 1, 28, 28)
            feature_output = self.features(dummy)
            flat_size = feature_output.view(1, -1).shape[1]

        self.classifier = nn.Sequential(
            nn.Flatten(),
            nn.Linear(flat_size, 128),
            l4_activation(),
            nn.Dropout(L4_DROPOUT),
            nn.Linear(128, 10)
        )

    def forward(self, x):
        return self.classifier(self.features(x))


def l4_count_params(model):
    return sum(p.numel() for p in model.parameters() if p.requires_grad)


def l4_run():
    model = ConfigurableCNN().to(l4_device)
    criterion = nn.CrossEntropyLoss()

    if L4_OPTIMIZER == "adam":
        optimizer = optim.Adam(model.parameters(), lr=L4_LR)
    elif L4_OPTIMIZER == "sgd":
        optimizer = optim.SGD(model.parameters(), lr=L4_LR, momentum=L4_MOMENTUM)
    else:
        raise ValueError("L4_OPTIMIZER must be 'adam' or 'sgd'")

    scheduler = None
    if L4_SCHEDULER == "cosine":
        scheduler = optim.lr_scheduler.CosineAnnealingLR(optimizer, T_max=L4_EPOCHS)

    train_losses, val_losses = [], []
    train_accs, val_accs = [], []
    epoch_times = []

    for epoch in range(L4_EPOCHS):
        start = time.time()
        model.train()
        loss_sum = 0.0
        correct = 0
        total = 0

        for x, y in l4_train_loader:
            x, y = x.to(l4_device), y.to(l4_device)
            optimizer.zero_grad()
            out = model(x)
            loss = criterion(out, y)
            loss.backward()
            optimizer.step()
            loss_sum += loss.item() * x.size(0)
            correct += (out.argmax(1) == y).sum().item()
            total += y.size(0)

        train_loss = loss_sum / total
        train_acc = correct / total

        model.eval()
        v_loss_sum = 0.0
        v_correct = 0
        v_total = 0
        with torch.no_grad():
            for x, y in l4_val_loader:
                x, y = x.to(l4_device), y.to(l4_device)
                out = model(x)
                loss = criterion(out, y)
                v_loss_sum += loss.item() * x.size(0)
                v_correct += (out.argmax(1) == y).sum().item()
                v_total += y.size(0)

        val_loss = v_loss_sum / v_total
        val_acc = v_correct / v_total

        if scheduler is not None:
            scheduler.step()

        train_losses.append(train_loss)
        val_losses.append(val_loss)
        train_accs.append(train_acc)
        val_accs.append(val_acc)
        epoch_times.append(time.time() - start)

        print(
            f"Epoch [{epoch+1}/{L4_EPOCHS}] "
            f"Train Loss: {train_loss:.4f} "
            f"Train Acc: {train_acc:.4f} "
            f"Val Loss: {val_loss:.4f} "
            f"Val Acc: {val_acc:.4f}"
        )

    model.eval()
    correct = 0
    total = 0
    test_loss = 0.0
    with torch.no_grad():
        for x, y in l4_test_loader:
            x, y = x.to(l4_device), y.to(l4_device)
            out = model(x)
            loss = criterion(out, y)
            test_loss += loss.item() * x.size(0)
            correct += (out.argmax(1) == y).sum().item()
            total += y.size(0)

    return model, {
        "train_loss": train_losses,
        "val_loss": val_losses,
        "train_acc": train_accs,
        "val_acc": val_accs,
        "test_accuracy": correct / total,
        "test_loss": test_loss / total,
        "time_per_epoch": np.mean(epoch_times),
        "parameters": l4_count_params(model)
    }

# AlexNet / VGG16 / ResNet18 / ResNet50

l5b_device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
L5B_EPOCHS = 3
L5B_BATCH_SIZE = 64
L5B_LR_HEAD = 1e-3
L5B_LR_BACKBONE = 1e-5

l5b_transform = transforms.Compose([
    transforms.Grayscale(num_output_channels=3),
    transforms.Resize((224, 224)),
    transforms.ToTensor(),
    transforms.Normalize(
        [0.485, 0.456, 0.406],
        [0.229, 0.224, 0.225]
    )
])

l5b_full_train = datasets.MNIST(
    root="./data", train=True, download=True, transform=l5b_transform
)
l5b_test_set = datasets.MNIST(
    root="./data", train=False, download=True, transform=l5b_transform
)

l5b_train_set, l5b_val_set = random_split(
    l5b_full_train,
    [54000, 6000],
    generator=torch.Generator().manual_seed(42)
)


def l5b_loaders(batch_size=64):
    return (
        DataLoader(l5b_train_set, batch_size=batch_size, shuffle=True),
        DataLoader(l5b_val_set, batch_size=batch_size, shuffle=False),
        DataLoader(l5b_test_set, batch_size=batch_size, shuffle=False)
    )


def l5b_create_model(name, pretrained=True):
    if name == "alexnet":
        weights = models.AlexNet_Weights.DEFAULT if pretrained else None
        model = models.alexnet(weights=weights)
        model.classifier[6] = nn.Linear(model.classifier[6].in_features, 10)
    elif name == "vgg16":
        weights = models.VGG16_Weights.DEFAULT if pretrained else None
        model = models.vgg16(weights=weights)
        model.classifier[6] = nn.Linear(model.classifier[6].in_features, 10)
    elif name == "resnet18":
        weights = models.ResNet18_Weights.DEFAULT if pretrained else None
        model = models.resnet18(weights=weights)
        model.fc = nn.Linear(model.fc.in_features, 10)
    elif name == "resnet50":
        weights = models.ResNet50_Weights.DEFAULT if pretrained else None
        model = models.resnet50(weights=weights)
        model.fc = nn.Linear(model.fc.in_features, 10)
    else:
        raise ValueError("Use alexnet, vgg16, resnet18 or resnet50")
    return model.to(l5b_device)


def l5b_freeze_feature_extraction(model, name):
    for p in model.parameters():
        p.requires_grad = False

    if name in ["alexnet", "vgg16"]:
        for p in model.classifier[6].parameters():
            p.requires_grad = True
    else:
        for p in model.fc.parameters():
            p.requires_grad = True


def l5b_unfreeze_all(model):
    for p in model.parameters():
        p.requires_grad = True


def l5b_partial_unfreeze_resnet(model, n):
    for p in model.parameters():
        p.requires_grad = False

    for p in model.fc.parameters():
        p.requires_grad = True

    if n == "all":
        l5b_unfreeze_all(model)
        return

    children = list(model.children())
    for module in children[-n:]:
        for p in module.parameters():
            p.requires_grad = True


def l5b_count_params(model):
    return sum(p.numel() for p in model.parameters() if p.requires_grad)


def l5b_train(model, train_loader, val_loader, test_loader,
              optimizer, epochs=3):
    criterion = nn.CrossEntropyLoss()
    train_losses = []
    val_accs = []
    epoch_times = []

    for epoch in range(epochs):
        start = time.time()
        model.train()
        loss_sum = 0.0
        correct = 0
        total = 0

        for x, y in train_loader:
            x, y = x.to(l5b_device), y.to(l5b_device)
            optimizer.zero_grad()
            out = model(x)
            loss = criterion(out, y)
            loss.backward()
            optimizer.step()
            loss_sum += loss.item() * x.size(0)
            correct += (out.argmax(1) == y).sum().item()
            total += y.size(0)

        model.eval()
        vcorrect = 0
        vtotal = 0
        with torch.no_grad():
            for x, y in val_loader:
                x, y = x.to(l5b_device), y.to(l5b_device)
                vcorrect += (model(x).argmax(1) == y).sum().item()
                vtotal += y.size(0)

        train_loss = loss_sum / total
        val_acc = vcorrect / vtotal
        train_losses.append(train_loss)
        val_accs.append(val_acc)
        epoch_times.append(time.time() - start)

        print(
            f"Epoch [{epoch+1}/{epochs}] "
            f"Loss: {train_loss:.4f} "
            f"Train Acc: {correct/total:.4f} "
            f"Val Acc: {val_acc:.4f}"
        )

    model.eval()
    correct = 0
    total = 0
    with torch.no_grad():
        for x, y in test_loader:
            x, y = x.to(l5b_device), y.to(l5b_device)
            correct += (model(x).argmax(1) == y).sum().item()
            total += y.size(0)

    return {
        "train_loss": train_losses,
        "val_acc": val_accs,
        "test_accuracy": correct / total,
        "parameters": l5b_count_params(model),
        "time_per_epoch": np.mean(epoch_times)
    }


def l5b_run_feature_extraction(name, batch_size=64):
    train_loader, val_loader, test_loader = l5b_loaders(batch_size)
    model = l5b_create_model(name, pretrained=True)
    l5b_freeze_feature_extraction(model, name)
    optimizer = optim.Adam(
        filter(lambda p: p.requires_grad, model.parameters()),
        lr=L5B_LR_HEAD
    )
    return l5b_train(
        model, train_loader, val_loader, test_loader,
        optimizer, L5B_EPOCHS
    )
