# 1. Import libraries
import numpy as np
import matplotlib.pyplot as plt
from tensorflow.keras.datasets import fashion_mnist
from tensorflow.keras.preprocessing.text import Tokenizer
from tensorflow.keras.preprocessing.sequence import pad_sequences
from tensorflow.keras.models import Model
from tensorflow.keras.layers import Input, Conv2D, MaxPooling2D
from tensorflow.keras.layers import GlobalAveragePooling2D, Dense, Embedding, LSTM, Add

# 2. Load dataset
(x_train, y_train), (x_test, y_test) = fashion_mnist.load_data() ##
x_train = x_train[:2000].astype("float32").reshape(-1,28,28,1) / 255.0 ##
x_test = x_test[:100].astype("float32").reshape(-1,28,28,1) / 255.0 ##

names = ["t-shirt","trouser","pullover","dress","coat", ##
         "sandal","shirt","sneaker","bag","boot"]

captions = ["startseq a photo of a " + names[i] + " endseq"
            for i in y_train[:2000]]

# 3. Tokenize captions
tok = Tokenizer()
tok.fit_on_texts(captions)
seq = tok.texts_to_sequences(captions)
vocab = len(tok.word_index) + 1
length = max(len(s) for s in seq)

# 4. Prepare next-word training data
images, words, targets = [], [], []
for img, s in zip(x_train, seq):
    for t in range(1, len(s)):
        images.append(img)
        words.append(pad_sequences([s[:t]], maxlen=length)[0])
        targets.append(s[t])

images = np.array(images)
words = np.array(words)
targets = np.array(targets)

# 5. CNN encoder
img_input = Input(shape=(28,28,1))  ##
x = Conv2D(32, 3, activation="relu", padding="same")(img_input)
x = MaxPooling2D()(x)
x = Conv2D(64, 3, activation="relu", padding="same")(x)
x = GlobalAveragePooling2D()(x)
img_features = Dense(128, activation="relu")(x)

# 6. LSTM decoder
word_input = Input(shape=(length,))
w = Embedding(vocab, 128, mask_zero=True)(word_input)
w = LSTM(128)(w)

# 7. Predict next word
z = Add()([img_features, w])
z = Dense(128, activation="relu")(z)
out = Dense(vocab, activation="softmax")(z)

model = Model([img_input, word_input], out)
model.compile(optimizer="adam",
              loss="sparse_categorical_crossentropy",
              metrics=["accuracy"])

# 8. Train
model.fit([images, words], targets,
          epochs=3, batch_size=64, verbose=1)

# 9. Generate caption
index_word = {i: w for w, i in tok.word_index.items()}

def caption_image(img):
    text = "startseq"

    for _ in range(length):
        s = tok.texts_to_sequences([text])[0]
        s = pad_sequences([s], maxlen=length)

        p = model.predict(
            [img.reshape(1,28,28,1), s], verbose=0
        )

        idx = np.argmax(p)
        word = index_word.get(idx, "")

        if not word or word == "endseq":
            break

        text += " " + word

    return text.replace("startseq", "").strip()

# 10. Display output
i = 0
print("Actual class:", names[int(y_test[i])])
print("Caption:", caption_image(x_test[i]))

plt.imshow(x_test[i].squeeze(), cmap="gray")
plt.axis("off")
plt.show()
